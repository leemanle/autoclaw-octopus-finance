from moviepy import *
import os
import numpy as np
from PIL import Image, ImageDraw, ImageFont

# 配置
OUTPUT_PATH = r"C:\Users\Administrator\Desktop\化工板块分析_抖音_1080P.mp4"

# 颜色定义
COLOR_BG = (26, 26, 46)
COLOR_WHITE = (255, 255, 255)
COLOR_RED = (255, 50, 50)
COLOR_ORANGE = (255, 140, 0)
COLOR_GOLD = (255, 215, 0)
COLOR_GREEN = (46, 139, 87)

print("Starting compliant video composition...")

def find_audio_file():
    import glob
    mp3_files = glob.glob(os.path.expandvars(r"%TEMP%\openclaw\**\*.mp3"), recursive=True)
    if mp3_files:
        print(f"Found audio: {mp3_files[0]}")
        return mp3_files[0]
    return None

def create_text_clip(text, duration, color=COLOR_WHITE, size=100):
    img = Image.new('RGB', (1920, 1080), color=COLOR_BG)
    draw = ImageDraw.Draw(img)
    
    font_size = size
    try:
        font = ImageFont.truetype("C:/Windows/Fonts/simhei.ttf", font_size)
    except:
        font = ImageFont.load_default()
    
    lines = text.split('\n')
    total_height = len(lines) * font_size
    y_start = (1080 - total_height) // 2
    
    for i, line in enumerate(lines):
        bbox = draw.textbbox((0, 0), line, font=font)
        text_width = bbox[2] - bbox[0]
        x = (1920 - text_width) // 2
        y = y_start + i * font_size * 1.2
        draw.text((x, y), line, fill=color, font=font)
    
    return ImageClip(np.array(img)).with_duration(duration)

try:
    audio_path = find_audio_file()
    
    print("Creating compliant clips...")
    # 注意：不提具体个股，只做行业分析
    
    # 分镜1
    clip1 = create_text_clip("化工板块\n强势崛起", 5, color=COLOR_RED, size=150)
    
    # 分镜2
    clip2 = create_text_clip("液氯涨300%\n扩散指数60.39%", 13, color=COLOR_ORANGE, size=120)
    
    # 分镜3
    clip3 = create_text_clip("油煤套利\n产业链机会", 20, color=COLOR_GOLD, size=100)
    
    # 分镜4：不提具体股票，只说板块
    clip4 = create_text_clip("煤化工板块\n景气度提升", 14, color=COLOR_GREEN, size=110)
    
    # 分镜5：添加风险提示
    clip5 = create_text_clip("市场有风险\n投资需谨慎", 8, color=COLOR_WHITE, size=100)
    
    print("Concatenating...")
    video_clip = concatenate_videoclips([clip1, clip2, clip3, clip4, clip5])
    
    if audio_path:
        print("Adding audio...")
        audio_clip = AudioFileClip(audio_path)
        if audio_clip.duration < 60:
            audio_clip = audio_clip.loop(duration=60)
        video_clip = video_clip.set_audio(audio_clip)
    
    print(f"Exporting to: {OUTPUT_PATH}")
    video_clip.write_videofile(
        OUTPUT_PATH,
        fps=30,
        codec='libx264',
        audio_codec='aac',
        bitrate='8000k',
        threads=4
    )
    
    print(f"\nCompliant video created!")
    print(f"Location: {OUTPUT_PATH}")
    print(f"Duration: {video_clip.duration}s")
    print(f"Resolution: 1920x1080")
    print(f"FPS: 30")
    print(f"Note: No specific stock codes mentioned")
    
except Exception as e:
